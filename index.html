function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // 取得現有第一列表頭 var lastCol = Math.max(sheet.getLastColumn(), 1); var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; // 欄位對應字典（確保網頁送出的 serial 與 contact 正確對應） var keyMapping = { 'timestamp': ['Timestamp', 'timestamp', '時間戳記'], 'status': ['Status', 'status', '狀態'], 'serial': ['Serial_No', 'serial', 'Serial', '序號', '流水號'], 'contact': ['Contact', 'contact', '聯絡方式', '電話信箱', 'D1'] }; var row = []; for (var i = 0; i < headers.length; i++) { var header = headers[i].toString().trim(); var val = ""; if (data[header] !== undefined) { val = data[header]; } else if (header === 'Serial_No' && data.serial !== undefined) { val = data.serial; } else if (header === 'Contact' && data.contact !== undefined) { val = data.contact; } else if (header === 'Contact' && data.D1 !== undefined && data.status === 'Reward_Registered') { val = data.D1; } else if (header === 'Status' && data.status !== undefined) { val = data.status; } else if (header === 'Timestamp' && data.timestamp !== undefined) { val = data.timestamp; } row.push(val); } sheet.appendRow(row); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput("ESG Survey Backend is Running."); }
